# Optional extras for the always-on daemon. Everything here degrades
# gracefully -- ASH runs without any of it, it just perceives less. Run
# `python ashd.py --check` to see exactly what each one buys you.

# --- always worth installing ---
psutil                  # system sensor, pid liveness, process names
Pillow                  # screenshot capture + perceptual dedup

# --- screen / window ---
mss                     # much faster screenshots than PIL.ImageGrab
rapidocr-onnxruntime    # reads the words on a captured screen. Without it,
#   screenshots are still *looked* at -- text vs video vs an empty desktop,
#   from numpy statistics, free -- but not read. onnxruntime only, no torch,
#   models are ~15MB. Costs ~6s of CPU per frame read on a 1080p screen, so
#   it runs on a worker thread and at most every sensors.screenshot.
#   read_text_every_s seconds (120 by default). Needs
#   sensors.screenshot.read_text = true, which is off with capture itself.
# Linux only, install via your package manager, not pip:
#   sudo apt install xdotool xprintidle   (active window + idle time)
# Windows and macOS need nothing extra for these.

# --- microphone ---
sounddevice             # audio capture (preferred)
webrtcvad               # voice activity detection; big accuracy win
faster-whisper          # LOCAL transcription -- keeps audio off the network
#   without faster-whisper, transcription falls back to Google's recognizer,
#   which uploads audio. PrivacyPolicy.require_local_asr blocks that by
#   default, so install this or the mic stays effectively text-less.

# --- camera ---
opencv-python           # webcam capture + motion gating

# --- PC actions (all dry-run by default) ---
pyautogui               # synthetic keyboard/mouse
pyperclip               # clipboard

# --- hardware / sensors / motors ---
pyserial                # serial sensors and motor control
